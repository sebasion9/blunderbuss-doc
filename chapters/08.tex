\chapter{Spis skrótów i symboli}

\begin{itemize}
\item[AST] drzewo składniowe (ang. \english{abstract syntax tree})
\item[API] interfejs programistyczny (ang. \english{application programming interface})
\item[RGB] struktura reprezentująca kolor za pomocą trzech składowych: czerwonej (Red), zielonej (Green) i niebieskiej (Blue)
\item[LSP] protokół serwera języka (ang. \english{language server protocol})
\item[PIE] plik wykonywalny niezależny od pozycji (ang. \english{position independent executable})
\end{itemize}
